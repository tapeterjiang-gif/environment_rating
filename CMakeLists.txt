cmake_minimum_required(VERSION 3.16)

# Can also be used directly as an ESP-IDF component.
if(ESP_PLATFORM)
    idf_component_register(SRCS "air_quality_rating.c" "thermal_comfort.c" "sound_light_rating.c" INCLUDE_DIRS ".")
    target_compile_features(${COMPONENT_LIB} PRIVATE c_std_99)
    return()
endif()

project(environment_rating LANGUAGES C)
add_library(air_quality_rating STATIC air_quality_rating.c)
target_include_directories(air_quality_rating PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(air_quality_rating PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)

include(CTest)
add_library(thermal_comfort STATIC thermal_comfort.c)
target_include_directories(thermal_comfort PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(thermal_comfort PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(UNIX)
    target_link_libraries(thermal_comfort PUBLIC m)
endif()
add_library(sound_light_rating STATIC sound_light_rating.c)
target_include_directories(sound_light_rating PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(sound_light_rating PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(UNIX)
    target_link_libraries(sound_light_rating PUBLIC m)
endif()
if(BUILD_TESTING)
    enable_language(CXX)
    add_executable(test_air_quality_rating tests/test_air_quality_rating.c)
    target_link_libraries(test_air_quality_rating PRIVATE air_quality_rating)
    if(UNIX)
        target_link_libraries(test_air_quality_rating PRIVATE m)
    endif()
    set_target_properties(test_air_quality_rating PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_test(NAME rating COMMAND test_air_quality_rating)

    add_executable(test_cpp_link tests/test_cpp_link.cpp)
    target_link_libraries(test_cpp_link PRIVATE air_quality_rating thermal_comfort sound_light_rating)
    set_target_properties(test_cpp_link PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    add_test(NAME cpp_link COMMAND test_cpp_link)
    add_executable(test_thermal_comfort tests/test_thermal_comfort.c)
    target_link_libraries(test_thermal_comfort PRIVATE thermal_comfort)
    set_target_properties(test_thermal_comfort PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_test(NAME thermal_comfort COMMAND test_thermal_comfort)
    add_executable(test_sound_light_rating tests/test_sound_light_rating.c)
    target_link_libraries(test_sound_light_rating PRIVATE sound_light_rating)
    set_target_properties(test_sound_light_rating PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    add_test(NAME sound_light_rating COMMAND test_sound_light_rating)
endif()
